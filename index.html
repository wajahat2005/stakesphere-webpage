from anthropic import AsyncAnthropic
from sphereagent.graph import Neo4jKnowledgeStore

# SphereAgent Claude 3.5 Orchestrator
client = AsyncAnthropic()
engine = SphereAgent(
    model="claude-3-5-sonnet-20241022",
    tools=[ASTAuditor(), CypherRunner()],
    temperature=0.0,
    max_tokens=8192
)

async def audit_codebase(repo_path: str):
    plan = await engine.plan_audit(repo_path)
    verified_patch = await engine.execute_and_verify(plan)
    return verified_patch
